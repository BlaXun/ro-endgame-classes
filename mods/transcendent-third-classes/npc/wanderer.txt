//===== transcendent-third-classes: Wanderer ======================================
//= The Gypsy's advancement.
//=
//= A Gypsy at base level 99 and job level 70, with no skill
//= points left, becomes a Wanderer here, beside the stock
//= Gypsy changer, which is untouched. There is no other way
//= to the class.
//=
//= The class is rAthena's transcendent Wanderer. It keeps every
//= skill it had (its tree inherits the first, second and
//= transcendent class) and adds its own; the job level starts
//= again at 1 and runs to 70; the base level runs to 175.
//==============================================================

valkyrie,55,56,3	script	Wanderer#Valkyrie	101,{
	.@n$ = "[Wanderer]";
	if (Class == Job_Wanderer_T) {
		mes .@n$;
		mes "You chose the dance. A whip for it is carried by the headless steed of the old roads, and the queen of the bees braids another under the full moon.";
		close;
	}
	if (Class != Job_Gypsy) {
		mes .@n$;
		mes "A Gypsy who has reached base level 99 and job level 70 goes on here, as a ^0000FFWanderer^000000.";
		close;
	}
	if (BaseLevel < 99 || JobLevel < 70) {
		mes .@n$;
		mes "Come back at base level 99 and job level 70.";
		close;
	}
	if (SkillPoint) {
		mes .@n$;
		mes "Spend your skill points first.";
		close;
	}
	mes .@n$;
	mes "You have walked the Gypsy's road to its end. A ^0000FFWanderer^000000 goes on from there.";
	next;
	mes .@n$;
	mes "You keep every skill you have. Your job level starts again at 1 and runs to 70, with skill points for the Wanderer's own skills.";
	mes "Your base level can rise to 175, and your stats past 99.";
	next;
	mes .@n$;
	mes "^FF0000This step cannot be undone.^000000";
	if (select("I take up the dance.:Not yet.") == 2)
		close;
	jobchange Job_Wanderer_T;
	mes .@n$;
	mes "Welcome, Wanderer.";
	close;
}
