//===== transcendent-third-classes: Mechanic ===================
//= The Whitesmith's advancement.
//=
//= A Whitesmith at base level 99 and job level 70, with no skill
//= points left, becomes a Mechanic here, beside the stock
//= Whitesmith changer, which is untouched. There is no other way
//= to the class.
//=
//= The class is rAthena's transcendent Mechanic. It keeps every
//= skill it had (its tree inherits the first, second and
//= transcendent class) and adds its own; the job level starts
//= again at 1 and runs to 70; the base level runs to 175.
//==============================================================

valkyrie,42,50,5	script	Mechanic#Valkyrie	731,{
	.@n$ = "[Mechanic]";
	if (Class == Job_Mechanic_T) {
		mes .@n$;
		mes "You chose the machine. The Mado Engineer in Einbroch will put you in one once you hold its licence, and the Black Marketeer beside him sells its fuel. Axes for it are carried by the creatures of fire in the volcano, by the queen of the ant hell and by the dragon of Abyss Lake.";
		close;
	}
	if (Class != Job_Whitesmith) {
		mes .@n$;
		mes "A Whitesmith who has reached base level 99 and job level 70 goes on here, as a ^0000FFMechanic^000000.";
		close;
	}
	if (BaseLevel < 99 || JobLevel < 70) {
		mes .@n$;
		mes "Come back at base level 99 and job level 70.";
		close;
	}
	if (SkillPoint) {
		mes .@n$;
		mes "Spend your skill points first.";
		close;
	}
	mes .@n$;
	mes "You have walked the Whitesmith's road to its end. A ^0000FFMechanic^000000 goes on from there.";
	next;
	mes .@n$;
	mes "You keep every skill you have. Your job level starts again at 1 and runs to 70, with skill points for the Mechanic's own skills.";
	mes "Your base level can rise to 175, and your stats past 99.";
	next;
	mes .@n$;
	mes "^FF0000This step cannot be undone.^000000";
	if (select("I take the machine.:Not yet.") == 2)
		close;
	jobchange Job_Mechanic_T;
	mes .@n$;
	mes "Welcome, Mechanic.";
	close;
}
