//===== transcendent-third-classes: Genetic ====================
//= The Creator's advancement.
//=
//= A Creator at base level 99 and job level 70, with no skill
//= points left, becomes a Genetic here, beside the stock
//= Creator changer, which is untouched. There is no other way
//= to the class.
//=
//= The class is rAthena's transcendent Genetic. It keeps every
//= skill it had (its tree inherits the first, second and
//= transcendent class) and adds its own; the job level starts
//= again at 1 and runs to 70; the base level runs to 175.
//==============================================================

valkyrie,55,50,3	script	Genetic#Valkyrie	122,{
	.@n$ = "[Genetic]";
	if (Class == Job_Genetic_T) {
		mes .@n$;
		mes "You chose the cart and the seed. Axes for it grow where the geographers root, and with the fallen bishop and the dark lord, who keep what they took from alchemists.";
		close;
	}
	if (Class != Job_Creator) {
		mes .@n$;
		mes "A Creator who has reached base level 99 and job level 70 goes on here, as a ^0000FFGenetic^000000.";
		close;
	}
	if (BaseLevel < 99 || JobLevel < 70) {
		mes .@n$;
		mes "Come back at base level 99 and job level 70.";
		close;
	}
	if (SkillPoint) {
		mes .@n$;
		mes "Spend your skill points first.";
		close;
	}
	mes .@n$;
	mes "You have walked the Creator's road to its end. A ^0000FFGenetic^000000 goes on from there.";
	next;
	mes .@n$;
	mes "You keep every skill you have. Your job level starts again at 1 and runs to 70, with skill points for the Genetic's own skills.";
	mes "Your base level can rise to 175, and your stats past 99.";
	next;
	mes .@n$;
	mes "^FF0000This step cannot be undone.^000000";
	if (select("I take the cart and the seed.:Not yet.") == 2)
		close;
	jobchange Job_Genetic_T;
	mes .@n$;
	mes "Welcome, Genetic.";
	close;
}
