//===== transcendent-third-classes: Sorcerer ===================
//= The Professor's advancement.
//=
//= A Professor at base level 99 and job level 70, with no skill
//= points left, becomes a Sorcerer here, beside the stock
//= Professor changer, which is untouched. There is no other way
//= to the class.
//=
//= The class is rAthena's transcendent Sorcerer. It keeps every
//= skill it had (its tree inherits the first, second and
//= transcendent class) and adds its own; the job level starts
//= again at 1 and runs to 70; the base level runs to 175.
//==============================================================

valkyrie,55,47,3	script	Sorcerer#Valkyrie	743,{
	.@n$ = "[Sorcerer]";
	if (Class == Job_Sorcerer_T) {
		mes .@n$;
		mes "You chose the spirits. Books for it are kept by the books that walk, and by the ice lord and the turtle general, who hoard what scholars lost. Your spirits answer to the stones of their element: Red Blood and a Flame Heart, Crystal Blue and Mystic Frozen, and so on.";
		close;
	}
	if (Class != Job_Professor) {
		mes .@n$;
		mes "A Professor who has reached base level 99 and job level 70 goes on here, as a ^0000FFSorcerer^000000.";
		close;
	}
	if (BaseLevel < 99 || JobLevel < 70) {
		mes .@n$;
		mes "Come back at base level 99 and job level 70.";
		close;
	}
	if (SkillPoint) {
		mes .@n$;
		mes "Spend your skill points first.";
		close;
	}
	mes .@n$;
	mes "You have walked the Professor's road to its end. A ^0000FFSorcerer^000000 goes on from there.";
	next;
	mes .@n$;
	mes "You keep every skill you have. Your job level starts again at 1 and runs to 70, with skill points for the Sorcerer's own skills.";
	mes "Your base level can rise to 175, and your stats past 99.";
	next;
	mes .@n$;
	mes "^FF0000This step cannot be undone.^000000";
	if (select("I call the spirits.:Not yet.") == 2)
		close;
	jobchange Job_Sorcerer_T;
	mes .@n$;
	mes "Welcome, Sorcerer.";
	close;
}
