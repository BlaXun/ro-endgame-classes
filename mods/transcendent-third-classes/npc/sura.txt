//===== transcendent-third-classes: Sura =======================
//= The Champion's advancement.
//=
//= A Champion at base level 99 and job level 70, with no skill
//= points left, becomes a Sura here, beside the stock
//= Champion changer, which is untouched. There is no other way
//= to the class.
//=
//= The class is rAthena's transcendent Sura. It keeps every
//= skill it had (its tree inherits the first, second and
//= transcendent class) and adds its own; the job level starts
//= again at 1 and runs to 70; the base level runs to 175.
//==============================================================

valkyrie,55,42,3	script	Sura#Valkyrie	52,{
	.@n$ = "[Sura]";
	if (Class == Job_Sura_T) {
		mes .@n$;
		mes "You chose the unbroken chain. Claws and knuckles made for it were carried by the owls of the old forest and by the tiger lords; take them from those who still keep them.";
		close;
	}
	if (Class != Job_Champion) {
		mes .@n$;
		mes "A Champion who has reached base level 99 and job level 70 goes on here, as a ^0000FFSura^000000.";
		close;
	}
	if (BaseLevel < 99 || JobLevel < 70) {
		mes .@n$;
		mes "Come back at base level 99 and job level 70.";
		close;
	}
	if (SkillPoint) {
		mes .@n$;
		mes "Spend your skill points first.";
		close;
	}
	mes .@n$;
	mes "You have walked the Champion's road to its end. A ^0000FFSura^000000 goes on from there.";
	next;
	mes .@n$;
	mes "You keep every skill you have. Your job level starts again at 1 and runs to 70, with skill points for the Sura's own skills.";
	mes "Your base level can rise to 175, and your stats past 99.";
	next;
	mes .@n$;
	mes "^FF0000This step cannot be undone.^000000";
	if (select("I take the endless chain.:Not yet.") == 2)
		close;
	jobchange Job_Sura_T;
	mes .@n$;
	mes "Welcome, Sura.";
	close;
}
