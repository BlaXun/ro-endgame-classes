//===== transcendent-third-classes: Ranger =====================
//= The Sniper's advancement.
//=
//= A Sniper at base level 99 and job level 70, with no skill
//= points left, becomes a Ranger here, beside the stock
//= Sniper changer, which is untouched. There is no other way
//= to the class.
//=
//= The class is rAthena's transcendent Ranger. It keeps every
//= skill it had (its tree inherits the first, second and
//= transcendent class) and adds its own; the job level starts
//= again at 1 and runs to 70; the base level runs to 175.
//==============================================================

valkyrie,42,55,5	script	Ranger#Valkyrie	727,{
	.@n$ = "[Ranger]";
	if (Class == Job_Ranger_T) {
		mes .@n$;
		if (!countitem(6124)) {
			getitem 6124, 1;
			mes "Lost your flute? Here is another. Your warg will not come while a falcon rides your arm.";
			close;
		}
		mes "You chose the wild. Bows for it hang in the lairs of the worms and demons of the old places, and in the dens of the wolf king and his kind.";
		close;
	}
	if (Class != Job_Sniper) {
		mes .@n$;
		mes "A Sniper who has reached base level 99 and job level 70 goes on here, as a ^0000FFRanger^000000.";
		close;
	}
	if (BaseLevel < 99 || JobLevel < 70) {
		mes .@n$;
		mes "Come back at base level 99 and job level 70.";
		close;
	}
	if (SkillPoint) {
		mes .@n$;
		mes "Spend your skill points first.";
		close;
	}
	mes .@n$;
	mes "You have walked the Sniper's road to its end. A ^0000FFRanger^000000 goes on from there.";
	next;
	mes .@n$;
	mes "You keep every skill you have. Your job level starts again at 1 and runs to 70, with skill points for the Ranger's own skills.";
	mes "Your base level can rise to 175, and your stats past 99.";
	next;
	mes .@n$;
	mes "^FF0000This step cannot be undone.^000000";
	if (select("I take the wild.:Not yet.") == 2)
		close;
	jobchange Job_Ranger_T;
	getitem 6124, 1;   // Wolf's Flute
	mes .@n$;
	mes "Welcome, Ranger. Take this flute: blow it, and your warg will come.";
	close;
}
