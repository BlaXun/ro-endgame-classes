//===== transcendent-third-classes: Royal Guard ===========
//= The Paladin's advancement.
//=
//= A Paladin at base level 99 and job level 70, with no skill
//= points left, becomes a Royal Guard here, beside the stock
//= Paladin changer, which is untouched. There is no other way
//= to the class.
//=
//= The class is rAthena's transcendent Royal Guard. It keeps every
//= skill it had (its tree inherits the first, second and
//= transcendent class) and adds its own; the job level starts
//= again at 1 and runs to 70; the base level runs to 175.
//==============================================================

valkyrie,55,39,3	script	Royal Guard#Valkyrie	752,{
	.@n$ = "[Royal Guard]";
	if (Class == Job_Royal_Guard_T) {
		mes .@n$;
		mes "You chose the shield. Spears and shields made for the Guard were lost with the knights of old; look for them among the dead who still keep their posts. The Peco Peco Breeder will give you your gryphon.";
		close;
	}
	if (Class != Job_Paladin) {
		mes .@n$;
		mes "A Paladin who has reached base level 99 and job level 70 goes on here, as a ^0000FFRoyal Guard^000000.";
		close;
	}
	if (BaseLevel < 99 || JobLevel < 70) {
		mes .@n$;
		mes "Come back at base level 99 and job level 70.";
		close;
	}
	if (SkillPoint) {
		mes .@n$;
		mes "Spend your skill points first.";
		close;
	}
	mes .@n$;
	mes "You have walked the Paladin's road to its end. A ^0000FFRoyal Guard^000000 goes on from there.";
	next;
	mes .@n$;
	mes "You keep every skill you have. Your job level starts again at 1 and runs to 70, with skill points for the Royal Guard's own skills.";
	mes "Your base level can rise to 175, and your stats past 99.";
	next;
	mes .@n$;
	mes "^FF0000This step cannot be undone.^000000";
	if (select("I take the shield.:Not yet.") == 2)
		close;
	jobchange Job_Royal_Guard_T;
	mes .@n$;
	mes "Welcome, Royal Guard.";
	close;
}
