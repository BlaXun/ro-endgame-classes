//===== transcendent-third-classes: Guillotine Cross ===========
//= The Assassin Cross's advancement.
//=
//= An Assassin Cross at base level 99 and job level 70, with no skill
//= points left, becomes a Guillotine Cross here, beside the stock
//= Assassin Cross changer, which is untouched. There is no other way
//= to the class.
//=
//= The class is rAthena's transcendent Guillotine Cross. It keeps every
//= skill it had (its tree inherits the first, second and
//= transcendent class) and adds its own; the job level starts
//= again at 1 and runs to 70; the base level runs to 175.
//==============================================================

valkyrie,42,58,5	script	Guillotine Cross#Valkyrie	725,{
	.@n$ = "[Guillotine Cross]";
	if (Class == Job_Guillotine_Cross_T) {
		mes .@n$;
		mes "You chose the guillotine. The guild's executioners carry blades made for our kind; take them from the desert's kings and their guards.";
		close;
	}
	if (Class != Job_Assassin_Cross) {
		mes .@n$;
		mes "An Assassin Cross who has reached base level 99 and job level 70 goes on here, as a ^0000FFGuillotine Cross^000000.";
		close;
	}
	if (BaseLevel < 99 || JobLevel < 70) {
		mes .@n$;
		mes "Come back at base level 99 and job level 70.";
		close;
	}
	if (SkillPoint) {
		mes .@n$;
		mes "Spend your skill points first.";
		close;
	}
	mes .@n$;
	mes "You have walked the Assassin Cross's road to its end. A ^0000FFGuillotine Cross^000000 goes on from there.";
	next;
	mes .@n$;
	mes "You keep every skill you have. Your job level starts again at 1 and runs to 70, with skill points for the Guillotine Cross's own skills.";
	mes "Your base level can rise to 175, and your stats past 99.";
	next;
	mes .@n$;
	mes "^FF0000This step cannot be undone.^000000";
	if (select("I take the guillotine.:Not yet.") == 2)
		close;
	jobchange Job_Guillotine_Cross_T;
	mes .@n$;
	mes "Welcome, Guillotine Cross.";
	close;
}
